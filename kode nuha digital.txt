            const inputEl = document.getElementById('chatInput');

            renderUserMessage(userMsg);
            inputEl.value = '';

                Pedoman Jawaban:

                4. Sertakan arahan kontak WhatsApp Petugas (+6285707175833) jika memerlukan pelayanan tatap muka/berkas resmi.
            `;

            const payload = {
                contents: [{ parts: [{ text: userQuery }] }],
                systemInstruction: { parts: [{ text: systemPrompt }] }
            };

            let delays = [1000, 2000, 4000];
            for (let i = 0; i <= delays.length; i++) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (response.ok) {

                        const candidate = data.candidates?.[0];
                        if (candidate && candidate.content?.parts?.[0]?.text) {
                            return candidate.content.parts[0].text;
                        }
                    }
                } catch (e) {
                    // Retry silently
                }

                if (i < delays.length) {
                    await new Promise(res => setTimeout(res, delays[i]));
                }
            }

            return "Terima kasih telah berkonsultasi. Untuk pertanyaan spesifik ini, Anda dapat menghubungi Petugas KUA Geger via WhatsApp di +6285707175833.";
        }

        function escapeHtml(text) {

        function formatMarkdown(text) {
            let formatted = escapeHtml(text);
            formatted = formatted.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
            formatted = formatted.replace(/\*(.*?)\*/g, '<em>$1</em>');
            formatted = formatted.replace(/\n/g, '<br>');
            return formatted;
